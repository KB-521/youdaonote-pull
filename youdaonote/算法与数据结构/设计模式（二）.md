5. 工厂模式：根据不同的参数产生不同类型的实例，这些实例都要有相同的行为；

```javascript
// 工厂模式实现新增节点
const creator = {
  text: text => document.createTextNode(text),
  img: url => {
    let img = document.createElement('img')
    img.src = url
    return img
  }
}
function createContent(type) { // 工厂函数
  let p = document.createElement('p'),
    content;
  var f = creator[type]
  if (f instanceof Function) {
    content = f.apply(null, Array.from(arguments).slice(1))
    p.appendChild(content);
    return p
  }
}
creator.link = (text, url) => { // 新增节点处理
  let a = document.createElement('a')
  a.href = url
  a.appendChild(document.createTextNode(text))
  return a
}

link = 'https://static.zhishuyao.com/img/fl_80x80.png'
document.body.appendChild(createContent('text', '图片'))
document.body.appendChild(createContent('img', link))
document.body.appendChild(createContent('link', '图片', link))
```
