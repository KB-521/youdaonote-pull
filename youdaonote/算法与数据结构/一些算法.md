进制装换：

```javascript
function baseConvert(input, from, to) {
  const charList = '0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIOJKLMNOPQRSTUVWXYZ'
  const charVal = function (char) {
    const code = char.charCodeAt(0)
    if (code >= 97) {
      return code - 97 + 10
    }
    if (code >= 65) {
      return code - 65 + 36
    }
    return code - 48
  }
  const input = String(input)
  let l = input.length,
    multiplier = 1,
    sum = 0
  for (let i = l - 1; i >= 0; i--) {
    sum += charVal(input[i]) * multiplier
    multiplier *= from
  }
  const stack = []
  while (sum > 0) {
    stack.push(charList[sum % to])
    sum = Math.floor(sum / to)
  }
  return stack.reverse().join("")
}
```
