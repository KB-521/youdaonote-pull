function bubbleSort(arr) {
    let l = arr.length
    for (let i = 0; i < l; i++) {
        let tmp
        for (let j = 0; j < l - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                tmp = arr[j + 1]
                arr[j + 1] = arr[j]
                arr[j] = tmp
            }
        }
    }
    return arr
}
```

function insertSort(arr) {
    let l = arr.length
    for (let i = 1; i < l; i++) {
        let item = arr[i]
        for (let j = 0; j < i; j++) {
            if (arr[j] > item) {
                arr.splice(j, 0, item)
                arr.splice(i + 1, 1)
                break
            }
        }
    }
    return arr
}
```

快速排序：

```javascript
// 取三个数中位值
function median(a, b, c) {
    if (a > b) {
        if (b > c) {
            return b
        } else {
            return a > c ? c : a
        }
    }
    if (c < a) {
        return a
    }
    return b > c ? c : b
}
function quickSort(arr) {
    const l = arr.length
    if (l <= 1) {
        return arr
    }
    const base = median(arr[0], arr[l-1], arr[Math.floor(l/2)])
    const left = [], right = [], middle = []
    for (var i = 0; i < l; i++) {
        const item = arr[i]
        if (item < base) {
            left.push(item)
        } else if (item > base) {
            right.push(item)
        } else {
            middle.push(item)
        }
    }
    return quickSort(left).concat(middle, quickSort(right))
}
```

// 分别排序、分别收集; 非比较、稳定; 时间复杂度 O(n*k);
function radixSort(arr) {
  let buckets = []
  let k = String(Math.max(...arr)).length
  for (let i = k - 1; i >= 0; i--) {
    for (let n of arr) {
      let dight = String(n).padStart(k, 0)[i]
      if (!buckets[dight]) {
        buckets[dight] = []
      }
      buckets[dight].push(n)
    }
    arr = buckets.reduce(
      (relust, bucket) => relust.concat(bucket),
      []
    )
    buckets = []
  }
  return arr
}
