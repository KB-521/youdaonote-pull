鼠标操作：

```javascript
.click(Fun); //鼠标左键单击
.dblclick(Fun); //鼠标左键双击
.contextmenu(function(e){
    return false
}); //取消鼠标右击默认事件
e.which; //判断你按下的是鼠标那个键(左键1、中键2、右键3)
.mouseenter(Fun)/mouseleave(Fun); //鼠标进入/离开(无冒泡阶段)
.mouseover(Fun)/mouseout(Fun); //鼠标进入/离开(有冒泡阶段)
.mousedown(Fun)/mouseup(Fun); //鼠标按下/弹起
.mousemove(Fun); //鼠标移动
```



焦点事件：

```javascript
.focus(Fun); //获得焦点触发
.blur(Fun); //失去焦点触发
.change(Fun); //失去焦点后里面的val值和上次不同时触发
```



键盘事件：

```javascript
.keydown(Fun); //键盘按下后触发(输入值前)
//e.which 返回对应的键值
.keyup(Fun); //键盘弹起后触发
.keypress(Fun); //键盘按下后触发(输入值前)
//e.which 返回对应的Ascll码值
```



滚动事件：

```javascript
.srcoll(Fun); //滚动指定元素时就会触发
```



Events：

```javascript
e.pageX; //获取鼠标离页面left值的距离
e.pageY; //获取鼠标离页面top值的距离
e.preventDefault(); //取消默认事件
e.stopPropagation(); //取消冒泡阶段
```



其他：

```javascript
.on(触发事件,Fun); //绑定事件方法
// 事件委托需要填入(触发事件,触发的子元素,选填的元素(存在data下),执行函数)
.off(触发事件); //解除事件方法
//氪选选填具体函数名称,来删除具体事件
.trigger(触发事件); //触发被选元素的指定事件类型
//可以用on自定义触发事件
.one(触发事件,Fun); //事件只能执行一次
```
