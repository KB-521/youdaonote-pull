abstract class Animal { // 抽象类:无法实例化，只能被继承
    private static age: number = 21 // 静态属性
    readonly gender:string = "man" // 只读、实例属性
    protected constructor(public name: string){
        this.name = name
    }
    public abstract sayHi(): string // 抽象方法:必须被子类发现
    static isAnimal(a): boolean{ // 静态方法:不需要实例化,直接通过类调用
        return a instanceof Animal
    }
}
//public: 默认,修饰的属性和方法都是共有的
//private: 无法被访问和继承
//protected: 无法被访问，但能被继承
class Cat extends Animal {
    constructor(name){
        super(name);
    }
    get name() { // 存取器
        return 'jack'
    }
    set name(value) { // 存取器  
        console.log('setter ', value)
    }
        public sayHi(){
        return `Meow, My name is ${this.name}`
    }
}

类与接口：

```typescript
// 不同类之间的相同特性可以用接口提取出来，再使用implements实现。
interface Alarm {
    alert(): void
}
interface LightableAlarm extends Alarm {
    lightOn(): void
    lightOff(): void
}
class Car implements Alarm, LightableAlarm { // 一个类可以实现多个接口
    alert() {}
    lightOn() {}
    lightOff() {}
}
// 接口可以继承接口, 接口也可以继承类。
// ts中在声明class Point时,除了创建一个Point的类之外，
// 也会创建一个Point类型(实例的类型)。所以接口继承接口和接口继承类没有本质的区别。
// 声明类所创建的类型不包含构造函数、静态属性和静态方法。
// 只包含其中的实例属性和实例方法。
```
