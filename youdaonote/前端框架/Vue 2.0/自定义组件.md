全局组件：

```javascript
Vue.component('my-component'{
    template: '<div></div>';
    data: function(){ //全局组件data是函数
        return{ }
    }
})
<my-component></my-component>使用自定义组件
```

局部组件：

```javascript
const vue1 = new Vue({
    el: 'dom',
    data:{ },
    components: {
        'my-component': {
            template: '<div></div>',
        }
    }
})
//局部组件只能在vue的节点中使用
//在其他的new Vue中用vue1.dataXX可以调用
```
