全局自定义指令：

```javascript
Vue.directive('自定义指令名称',{
    bind(el,binding,vnode){},//第一次绑定时触发
    update(el,binding,vnode,oldVnode){},//虚拟dom更新时触发
    componentUpdated(el,binding,vnode,oldVnode){},//虚拟dom及其子虚拟dom全部更新后触发
    unbind(el,binding,vnode){}//解绑时触发
})
//el: 绑定的DOM元素
//binding: 绑定指令时传的一些参数
//vnode: 更新后的虚拟dom
//oldVnode: 更新前的虚拟dom
```



局部自定义组件：

```javascript
export default {
  directives:{
      '自定义指令名称': {与全局指令一样}
  }  
}
```
