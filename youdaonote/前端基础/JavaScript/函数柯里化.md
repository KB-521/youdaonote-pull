> 柯里化：就是将接受多个参数的函数转变成接受一个单一参数(最初函数的第一个参数)的函数，并且返回接受余下参数而且返回结果的新函数的技术。



```javascript
function add(){
    var _args = Array.prototype.slice.call(arguments);
    
    var _adder = function() {
        _args.push(...arguments)
        return _adder
    }

    _adder.toString = function() {
        return _args.reduce(function (a, b){
            return a + b
        })
    }
    return _adder
}
add(1)(2)(3) // 6
add(1, 2, 3)(4) //10
add(1, 2, 3)(4, 5) // 15
```
