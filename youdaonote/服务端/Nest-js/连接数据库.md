安装：

```typescript
yarn add @nestjs/typeorm typeorm mysql
```



引入：

```typescript
//app.module.ts
import { TypeOrmModule } from '@nestjs/typeorm'
@Module({
	imports: [TypeOrmModule.forRoot(), ]
  ...
})
```



配置：

```typescript
// 根目录下新建ormconfig.js
module.exports = {
  "type": "数据库类型(如: mysql)",
  "host": "数据库连接地址",
  "port": 端口,
  "username": "用户名",
  "password": "密码",
  "database": "数据库名",
  entities: process.env.NODE_ENV == 'ts-node' ? ["src/entities/**/**.entity{.ts,.js}"] : ["dist/entities/**/**.entity{.ts,.js}"], 
   //表示扫描到的实例类的位置，src或dist下面的/entities文件夹下面的任意.ts或.js文件
  "synchronize": true //是否同步
}
```



debug配置：

vscode的默认配置中，会屡次读取dist缓存目录而不重新编译产生bug，需要设置

```typescript
// 新建.vscode/setting.json文件
{
    "git.ignoreLimitWarning": true,
    "typescript.preferences.importModuleSpecifier": "relative",
    "search.followSymlinks": false,
    "debug.node.autoAttach": "off",
    "typescript.tsdk": "node_modules\\typescript\\lib",
    "files.autoSave": "onFocusChange",
    "editor.formatOnSave": true,
    "editor.codeActionsOnSave": { "source.fixAll.eslint": true },
    "eslint.enable": true
}
```
